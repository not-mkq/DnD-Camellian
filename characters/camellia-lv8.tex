% 1. 定义一个内部函数，将 1-9 转换为对应的英文序数词
\cs_new:Npn \__spell_num_to_word:n #1
{
  \int_case:nn { #1 }
  {
    { 1 } { First }
    { 2 } { Second }
    { 3 } { Third }
    { 4 } { Fourth }
    { 5 } { Fifth }
    { 6 } { Sixth }
    { 7 } { Seventh }
    { 8 } { Eighth }
    { 9 } { Ninth }
  }
}

% 2. 定义用户宏 \spell，接收 6 个参数
\NewDocumentCommand { \spellMKQ } { m m m m m m }
{
  % 构造并执行第一个控制序列，例如 \FirstLevelSpellSlot[#2]{#3(#4) #5}
  \use:c { \__spell_num_to_word:n {#1} LevelSpellSlot #2 } 
  { 
    #3 ~ ( #4 ) 
    % 检查 suffix 是否为空。如果不为空，则在前面加一个空格
    \tl_if_blank:nF { #5 } { ~ #5 } 
  }
  
  % 构造并执行第二个控制序列，以及后续的原样参数
  \use:c { \__spell_num_to_word:n {#1} LevelSpellSlot #2 Prepared } { #6 }
}
